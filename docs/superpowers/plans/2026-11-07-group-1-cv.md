```ts
import type { ErrorCode } from "../cv/types";

// Mistake memory (doc item b): per letter, how often each error happened and when last. Browser only.
export type Mistakes = Record<string, Partial<Record<ErrorCode, { count: number; lastAt: number }>>>;

const KEY = "handshake.mistakes.v1";

export function loadMistakes(): Mistakes {
  try {
    return JSON.parse(localStorage.getItem(KEY) ?? "{}");
  } catch {
    return {};
  }
}

export function saveMistakes(m: Mistakes) {
  try {
    localStorage.setItem(KEY, JSON.stringify(m));
  } catch {
    /* session only */
  }
}

export function recordMistake(m: Mistakes, letter: string, code: ErrorCode, now = Date.now()): Mistakes {
  const count = (m[letter]?.[code]?.count ?? 0) + 1;
  const next = { ...m, [letter]: { ...m[letter], [code]: { count, lastAt: now } } };
  saveMistakes(next);
  return next;
}

export const mistakeCount = (m: Mistakes, letter: string, code: ErrorCode) => m[letter]?.[code]?.count ?? 0;

export function weakFingers(m: Mistakes, limit = 3) {
  const by = new Map<string, number>();
  for (const codes of Object.values(m))
    for (const [code, s] of Object.entries(codes)) {
      const finger = code.split(":")[0];
      by.set(finger, (by.get(finger) ?? 0) + (s?.count ?? 0));
    }
  return [...by].map(([finger, count]) => ({ finger, count })).sort((a, b) => b.count - a.count).slice(0, limit);
}
```

```ts
/** Runs the model on the current frame. With two hands visible, returns the signer's dominant hand. */
export function detectHand(lm: HandLandmarker, video: HTMLVideoElement | HTMLCanvasElement, now: number, dominant: "left" | "right" = "right"): HandFrame | null {
  const result = lm.detectForVideo(video, now);
  if (!result.landmarks.length) return null;
  const labels = result.handedness.map((h) => (h[0]?.categoryName === "Left" ? "Left" : "Right") as "Left" | "Right");

  return { landmarks: result.landmarks[i], worldLandmarks: result.worldLandmarks?.[i], handedness: labels[i] };
}
```

```ts
import { useEffect, useMemo, useRef, useState } from "react";
import { useHandTracking } from "./useHandTracking";
import { extractFeatures } from "./features";
import { loadTemplates, meanTemplate } from "./classifier";
import { Recognizer, candidatesFor } from "./recognizer";
import { VoteBuffer } from "./temporal";
import { HoldConfirm } from "./holdConfirm";
import { getHandFeedback } from "./feedback";
import type { FingerFeedback, HandFrame, Prediction, SignDetection, Template } from "./types";

const MISS_MS = 4000;
const NONE: Prediction = { letter: null, confidence: 0, distance: Infinity };

export type LetterCheckOptions = {
  leftHanded?: boolean;
  holdMs?: number;
  enabled?: boolean;
  onMatch?: (d: SignDetection) => void;
  onMiss?: (feedback: FingerFeedback[], missCount: number) => void;
  onFrame?: (hand: HandFrame | null, pred: Prediction, now: number) => void;
};

/** A hand touching the frame edge gives bad landmarks: treat it as no hand. */
export const inFrame = (h: HandFrame) => h.landmarks.every((p) => p.x > 0.01 && p.x < 0.99 && p.y > 0.01 && p.y < 0.99);

// One recognizer for the whole app, so the MLP loads once.
let engine: { templates: Template[]; recognizer: Recognizer } | null = null;
function getEngine() {
  if (!engine) {
    const templates = loadTemplates();
    engine = { templates, recognizer: new Recognizer(templates) };
  }
  return engine;
}

export function useLetterCheck(target: string, lessonLetters: string[], opts: LetterCheckOptions = {}) {
  const { templates, recognizer } = getEngine();
  const [progress, setProgress] = useState(0);
  const [prediction, setPrediction] = useState<Prediction | null>(null);
  const [missCount, setMissCount] = useState(0);
  const [hint, setHint] = useState("");
  const hold = useRef(new HoldConfirm(opts.holdMs ?? 800));
  const votes = useRef(new VoteBuffer(10));
  const since = useRef(performance.now());
  const misses = useRef(0);
  const lastHand = useRef<HandFrame | null>(null);
  const o = useRef(opts);
  o.current = opts;
  const lettersKey = lessonLetters.join("");
  const candidates = useMemo(() => candidatesFor(target, lessonLetters), [target, lettersKey]); // eslint-disable-line react-hooks/exhaustive-deps

  function reset() {
    hold.current.reset();
    votes.current.reset();
    since.current = performance.now();
    misses.current = 0;
    setMissCount(0);
    setProgress(0);
  }
  useEffect(reset, [target]);

  const cam = useHandTracking(
    (raw, now) => {
      const hand = raw && inFrame(raw) ? raw : null;
      lastHand.current = hand;
      setHint(raw && !hand ? "Move your whole hand into the frame" : "");
      if (o.current.enabled === false || !target) {
        hold.current.reset();
        since.current = now;
        return;
      }
      if (!hand) {
        // No hand = start over: a fresh full hold is needed, and time without a hand isn't a miss.
        setPrediction(null);
        setProgress(0);
        votes.current.reset();
        hold.current.reset();
        since.current = now;
        o.current.onFrame?.(null, NONE, now);
        return;
      }
      const features = extractFeatures(hand);
      const p = votes.current.push(recognizer.predict(features, candidates));
      setPrediction(p);
      o.current.onFrame?.(hand, p, now);
      const d = hold.current.update(p, now);
      setProgress(hold.current.progress(now));
      if (d?.letter === target) {
        reset();
        o.current.onMatch?.(d);
        return;
      }
      if (now - since.current > MISS_MS) {
        since.current = now;
        misses.current += 1;
        setMissCount(misses.current);
        const mean = meanTemplate(target, templates);
        o.current.onMiss?.(mean ? getHandFeedback(features, mean) : [], misses.current);
      }
    },
    { dominant: opts.leftHanded ? "left" : "right" },
  );

  return { cam, progress, prediction, missCount, hint, templates, reset, snapshot: () => cam.snapshot(lastHand.current) };
}
```

```tsx
import { Suspense, lazy, useEffect, useMemo, useRef, useState } from "react";
import { isPracticeable, meanPose3D, useLetterCheck } from "../cv";
import { buildCurriculum, flatLessons, type Step } from "../lib/curriculum";
import { completeLesson, dueItems, isUnlocked, loadProgress, recordAttempt, type Progress } from "../lib/progress";
import { loadMistakes, mistakeCount, recordMistake } from "../lib/mistakes";
import { DEMO_PROFILE, loadProfile } from "../lib/profile";
import { SEND_SNAPSHOT, getCoachTip } from "../lib/api";
import { LessonView } from "../components/LessonView";
import { Camera } from "../components/Camera";
import { celebrate } from "../components/celebrate";

const HandModel3D = lazy(() => import("../components/HandModel3D"));

/** Spec §1: "Matches our recording" until a fluent signer has checked the templates (Task 11 Step 2). */
export const MATCH_WORDING = "Matches our recording";

// A lesson's steps flattened into one item per thing the learner does.
export type Item =
  | { kind: "learn"; letter: string; referenceOnly: boolean }
  | { kind: "check"; letter: string; showReference: boolean; word?: { letters: string[]; index: number } }
  | { kind: "other"; step: Step }; // sign / sentence / scene: group-1 Tasks 10 and 12

export function expand(steps: Step[], _progress: Progress): Item[] {
  const out: Item[] = [];
  for (const s of steps) {
    if (s.kind === "learn") out.push({ kind: "learn", letter: s.letter, referenceOnly: !isPracticeable(s.letter) });
    else if (s.kind === "practice") {
      if (isPracticeable(s.letter)) for (let i = 0; i < s.reps; i++) out.push({ kind: "check", letter: s.letter, showReference: true });
    } else if (s.kind === "spell") {
      const letters = [...s.word];
      letters.forEach((l, index) => isPracticeable(l) && out.push({ kind: "check", letter: l, showReference: s.showReference, word: { letters, index } }));
    } else if (s.kind === "contrast" || s.kind === "review") continue; // added in M2 (Task 9)
    else out.push({ kind: "other", step: s });
  }
  return out;
}

export function LessonPlayer({ lessonId }: { lessonId: string }) {
  const profile = loadProfile() ?? DEMO_PROFILE;
  const units = useMemo(() => buildCurriculum(profile), []); // eslint-disable-line react-hooks/exhaustive-deps
  const lesson = useMemo(() => flatLessons(units).find((l) => l.id === lessonId), [units, lessonId]);
  const progress = useRef(loadProgress()); // a ref, not state: recordAttempt mutates, and StrictMode would double-apply updaters
  const items = useMemo(() => (lesson ? expand(lesson.steps, progress.current) : []), [lesson]);
  const [i, setI] = useState(0);
  const lessonLetters = useMemo(() => [...new Set(items.flatMap((x) => (x.kind === "check" ? [x.letter] : [])))], [items]);

  function next() {
    if (i + 1 < items.length) return setI(i + 1);
    progress.current = completeLesson(progress.current, lesson!.id, lesson!.xp);
    location.hash = `#/complete/${lesson!.id}`;
  }

  if (!lesson) return <div className="page"><p>Lesson not found. <a href="#/">Back to your path</a></p></div>;
  // Same baseline: no skipping ahead by URL. "Skip" inside a lesson passes an item without counting it as a match.
  if (!isUnlocked(units, progress.current, lesson.id)) return <div className="page"><p>Locked: finish the lessons before it first. <a href="#/">Back to your path</a></p></div>;
  const item = items[i];
  if (!item) return <div className="page"><p>Nothing to practise here yet.</p><button onClick={next}>Continue</button></div>;
  const stepLabel = `Step ${i + 1} of ${items.length}`;

  // ONE camera at a time: LetterItem stays mounted across consecutive letter items (no key), so the camera never restarts.
  if (item.kind === "other")
    return <div className="page"><p>Coming in a later milestone: {item.step.kind}</p><button onClick={next}>Continue</button></div>;
  return <LetterItem index={i} item={item} lessonLetters={lessonLetters} leftHanded={profile.leftHanded} stepLabel={stepLabel} progress={progress} onNext={next} />;
}

type LetterItemProps = {
  index: number;
  item: Exclude<Item, { kind: "other" }>;
  lessonLetters: string[];
  leftHanded: boolean;
  stepLabel: string;
  progress: { current: Progress };
  onNext: () => void;
};

function LetterItem({ index, item, lessonLetters, leftHanded, stepLabel, progress, onNext }: LetterItemProps) {
  const [landmarkTip, setLandmarkTip] = useState("");
  const [coach, setCoach] = useState({ tip: "", cached: false });
  const [sideBySide, setSideBySide] = useState(false);
  const [status, setStatus] = useState("");
  const asking = useRef(false);
  const gen = useRef(0); // bumps per item: a late coach reply for the previous item is dropped
  const target = item.letter;

  useEffect(() => {
    gen.current++;
    asking.current = false;
    setLandmarkTip("");
    setCoach({ tip: "", cached: false });
    setSideBySide(false);
  }, [index]);

  const check = useLetterCheck(target, lessonLetters, {
    leftHanded,
    enabled: item.kind === "check",
    onMatch: (d) => {
      progress.current = recordAttempt(progress.current, d.letter, true);
      setStatus(`${MATCH_WORDING}: ${d.letter}`);
      celebrate();
      onNext();
    },
    onMiss: (feedback) => {
      progress.current = recordAttempt(progress.current, target, false);
      const first = feedback[0];
      setStatus(`Not matching ${target} yet`);
      setLandmarkTip(first ? `Your ${first.finger} ${first.issue}.` : "Compare your hand with the reference and hold it still.");
      if (!first) return;
      const repeat = mistakeCount(recordMistake(loadMistakes(), target, first.code), target, first.code);
      setSideBySide(repeat >= 3);
      if (asking.current) return;
      asking.current = true;
      const g = gen.current;
      getCoachTip(target, feedback, repeat, SEND_SNAPSHOT ? check.snapshot() : undefined)
        .then((r) => g === gen.current && setCoach({ tip: r.tip, cached: !!r.cached }))
        .catch(() => {}) // the landmark tip is already on screen
        .finally(() => g === gen.current && (asking.current = false));
    },
  });

  const camera = (
    <>
      <Camera {...check.cam} progress={check.progress} />
      {check.hint && <p role="status">{check.hint}</p>}
    </>
  );
  const reference = (
    <Suspense fallback={<div className="hand3d empty">Loading 3D hand…</div>}>
      <HandModel3D pose={meanPose3D(target, check.templates, leftHanded)} letter={target} />
    </Suspense>
  );
  const common = { target, reference, camera, landmarkTip, coachTip: coach.tip, coachCached: coach.cached, sideBySide, stepLabel };

  if (item.kind === "learn")
    return (
      <LessonView {...common} title={`Learn ${item.letter}`} showReference progress={0}
        status={item.referenceOnly ? "Reference only, coming later" : `This is ${item.letter}. Try it, then continue.`}
        actionLabel="Continue" onAction={onNext} />
    );

  return (
    <LessonView {...common} title={item.word ? `Spell ${item.word.letters.join("")}` : `Sign ${item.letter}`} word={item.word}
      showReference={item.showReference || sideBySide} progress={check.progress} status={status} actionLabel="Skip" onAction={onNext} />
  );
}
```

```tsx
import { useMemo, useState, type ReactNode } from "react";
import { SIGNS, SignChecker, isPracticeable, useLetterCheck, zoneOf } from "../cv";
import { Camera } from "../components/Camera";

/** Cut rule (H14): false → every sign is self-checked with "I signed it". */
export const SIGNS_AUTO = true;

/** `token` = index of the original token, so "fs:RYAN" letters all highlight the same chip. */
export type RunUnit = ({ kind: "sign"; id: string } | { kind: "letter"; letter: string }) & { token: number };

/** "fs:LILY" → fingerspelled letters (practiceable only); anything else is a SIGNS id. */
export const unitsOf = (tokens: string[]): RunUnit[] =>
  tokens.flatMap((t, token): RunUnit[] =>
    t.startsWith("fs:") ? [...t.slice(3)].filter(isPracticeable).map((letter) => ({ kind: "letter", letter, token })) : [{ kind: "sign", id: t, token }],
  );

type RenderArgs = { camera: ReactNode; prompt: string; index: number; token: number; units: RunUnit[]; selfCheck?: () => void };

export function TokenRunner(props: { tokens: string[]; leftHanded: boolean; onDone: () => void; render: (a: RenderArgs) => ReactNode }) {
  const units = useMemo(() => unitsOf(props.tokens), [props.tokens.join(" ")]); // eslint-disable-line react-hooks/exhaustive-deps
  const [k, setK] = useState(0);
  const [msg, setMsg] = useState("");
  const [zone, setZone] = useState("");
  const u = units[k];
  const def = u?.kind === "sign" ? SIGNS[u.id] : undefined;
  const auto = u?.kind === "letter" || (SIGNS_AUTO && def?.grading === "assisted");
  const checker = useMemo(() => (def ? new SignChecker(def) : null), [def]);
  const advance = () => {
    setMsg("");
    checker?.reset(); // the same sign can come up again (e.g. YOU in consecutive scene turns)
    if (k + 1 < units.length) setK(k + 1);
    else {
      setK(0);
      props.onDone();
    }
  };

  const check = useLetterCheck(u?.kind === "letter" ? u.letter : def?.handshape ?? "", [], {
    leftHanded: props.leftHanded,
    enabled: !!u && auto,
    onMatch: () => u?.kind === "letter" && advance(),
    onFrame: (hand, pred, now) => {
      if (!checker || !auto) return;
      if (hand) setZone(zoneOf(hand));
      const r = checker.update(hand, pred, now);
      setMsg(r.message);
      if (r.stage === "done") advance();
    },
  });

  if (!u) return null;
  const prompt =
    u.kind === "letter" ? `Fingerspell ${u.letter}` : !def ? `Sign ${u.id}` : auto ? msg || def.howTo : `${def.howTo} Then press "I signed it".`;
  const camera = (
    <>
      <Camera {...check.cam} progress={check.progress} />
      <p className="muted">debug: zone {zone || "–"} · seeing {check.prediction?.letter ?? "–"}</p>
    </>
  );
  return <>{props.render({ camera, prompt, index: k, token: u.token, units, selfCheck: auto ? undefined : advance })}</>;
}
```
